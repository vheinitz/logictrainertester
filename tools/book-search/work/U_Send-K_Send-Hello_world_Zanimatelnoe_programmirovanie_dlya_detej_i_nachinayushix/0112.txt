 import pygame, sys, random

 skier_images = ["skier_down.png", "skier_right1.png",
                 "skier_right2.png", "skier_left2.png",
                 "skier_left1.png"]

 class SkierClass(pygame.sprite.Sprite):
     def __init__(self):
         pygame.sprite.Sprite.__init__(self)
         self.image = pygame.image.load("skier_down.png")            Создаем лыжника
         self.rect = self.image.get_rect()
         self.rect.center = [320, 100]
         self.angle = 0

      def turn(self, direction):
          self.angle = self.angle + direction
          if self.angle < -2: self.angle = -2
          if self.angle > 2: self.angle = 2
          center = self.rect.center                                Поворачи-
          self.image = pygame.image.load(skier_images[self.angle]) ваем лыж-
                                                                   ника
          self.rect = self.image.get_rect()
          self.rect.center = center
          speed = [self.angle, 6 - abs(self.angle) * 2]
          return speed

      def move(self, speed):
                                                                        Двигаем
          self.rect.centerx = self.rect.centerx + speed[0]              лыжника влево
          if self.rect.centerx < 20: self.rect.centerx = 20             и вправо
          if self.rect.centerx > 620: self.rect.centerx = 620

 class ObstacleClass(pygame.sprite.Sprite):
