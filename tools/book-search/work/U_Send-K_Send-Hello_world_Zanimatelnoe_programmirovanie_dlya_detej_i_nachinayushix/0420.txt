 if self.robot.rotation == targetOrientation:

Иначе нужно определить, в какую сторону вам следует повернуться. Первым делом счи-

 else:
     leftTurnsNeeded = (self.robot.rotation - targetOrientation) % 4

Затем приходит время собственно поворота. Если для корректной ориентации нам тре-
буется более двух поворотов влево, имеет смысл один раз повернуться вправо:

 if leftTurnsNeeded <= 2:
     self.robot.turnLeft()
 else:
     self.robot.turnRight()

 class AI:
     def __init__(self):
         pass
     def turn(self):
         if self.robot.lookInFront() == "bot":
             self.robot.attack()
         else:
             self.goTowards(self.robot.locateEnemy()[0])
     def goTowards(self,enemyLocation):
         myLocation = self.robot.position
         delta = (enemyLocation[0]-myLocation[0],
                 enemyLocation[1]-myLocation[1])
         if abs(delta[0]) > abs(delta[1]):
             if delta[0] < 0:              Лицом влево
                 targetOrientation = 3
             else:                               Лицом вправо
                 targetOrientation = 1
         else:
             if delta[1] < 0:               Лицом вверх
                 targetOrientation = 0
             else:                         Лицом вниз
                 targetOrientation = 2
