 running = True                        Это переменная
                                       скорости speed          Помещаем код вывода
 while running:
                                                               мяча на экран внутрь
     for event in pygame.event.get():                          этого цикла while
         if event.type == pygame.QUIT:
         running = False

     pygame.time.delay(20)
     pygame.draw.rect(screen, [255,255,255], [x, y, 90, 90], 0)
     x = x + x_speed
     if x > screen.get_width() - 90 or x < 0:                Когда мяч
         x_speed = - x_speed                                 достигает
     screen.blit(my_ball, [x, y])                            края окна…
     pygame.display.flip()            …меняем направление его
 pygame.quit()                        движения, сменив знак скорости


За отскоки мяча от краев окна отвечают строки 19 и 20. В строке 19 (if x > screen.
get_width() - 90 or x < 0:) мы определяем, достиг ли мяч края окна, и в случае по-
ложительного результата проверки в строке 20 меняем направление его движения
(x_speed = - x_speed).
